URP / HDRP Setup Steps
  1. Drag and drop the downloaded PNG into your Unity Project window.
  2. In Texture Inspector: uncheck sRGB (Color Texture).
  3. Set Filter Mode = Bilinear, Compression = None, Wrap Mode = Clamp.
  4. URP: In your scene, select Global Volume → Add Override → Post-processing → Color Lookup.
  5. Assign the baked PNG into the Lookup Texture slot and set Contribution = 1.
  6. HDRP: In your Volume → Add Color Lookup override → assign texture.
  7. Tonemapping stays in Unity (ACES / Neutral); this LUT cleanly drives aesthetic color grading without any runtime allocations.
Zero-Alloc C# Volume Controller

Smoothly blend between LUT profiles dynamically at runtime with zero GC allocations.

using UnityEngine; using UnityEngine.Rendering; using UnityEngine.Rendering.Universal; public sealed class LutVolumeController : MonoBehaviour { [SerializeField] private Volume volume; [SerializeField] private Texture2D defaultLut; private ColorLookup colorLookup; private void Awake() { if (volume == null) volume = GetComponent(); if (volume != null && volume.profile != null) { volume.profile.TryGet(out colorLookup); } if (colorLookup != null && defaultLut != null) { colorLookup.texture.value = defaultLut; colorLookup.contribution.value = 1f; } } public void SetLut(Texture2D newLut, float weight = 1f) { if (colorLookup == null) return; colorLookup.texture.value = newLut; colorLookup.contribution.value = Mathf.Clamp01(weight); } }
